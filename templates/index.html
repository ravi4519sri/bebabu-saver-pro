function download(format) { const url = urlInput.value.trim(); if (!url) { showMessage('❌ Please enter a valid URL!', 'error'); return; } let platform = 'unknown'; if (url.includes('youtube.com') || url.includes('youtu.be')) { platform = 'youtube'; } else if (url.includes('starmaker')) { platform = 'starmaker'; } else { showMessage('❌ Only YouTube or StarMaker URLs are supported!', 'error'); return; } mp3Btn.disabled = true; mp4Btn.disabled = true; showMessage(`⏳ ${platform.toUpperCase()} ${format.toUpperCase()} download starting...`, 'info', platform); fetch('/download', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ url: url, format_type: format }) }) .then(response => response.json()) .then(data => { if (data.success) { showMessage(`✅ ${platform.toUpperCase()} ${format.toUpperCase()} download completed! File downloading now...`, 'success', platform); // Auto-download the file const downloadId = data.download_id; setTimeout(() => { window.location.href = `/get_file/${downloadId}`; }, 1000); urlInput.value = ''; } else { showMessage(`❌ ${data.message}`, 'error'); } }) .catch(error => { showMessage(`❌ Error: ${error.message}`, 'error'); }) .finally(() => { setTimeout(() => { mp3Btn.disabled = false; mp4Btn.disabled = false; }, 2000); }); }
